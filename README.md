## استفاده

```php
use Tavaan\Engine\Filesystem\FilesystemManager;
use Tavaan\Engine\Filesystem\Drivers\LocalFilesystem;
use Tavaan\Engine\Filesystem\Drivers\MemoryFilesystem;

// Manager با چند disk
$m = new FilesystemManager([
    'default' => 'local',
    'disks' => [
        'local' => ['driver' => 'local', 'root' => 'storage/app'],
        'public' => ['driver' => 'local', 'root' => 'storage/public'],
        'memory' => ['driver' => 'memory'],
    ],
]);

// عملیات روی disk پیش‌فرض
$m->put('file.txt', 'content');
$m->get('file.txt');
$m->exists('file.txt');
$m->delete('file.txt');
$m->copy('a.txt', 'b.txt');
$m->move('a.txt', 'b.txt');
$m->size('file.txt');
$m->mimeType('file.jpg');

$m->append('log.txt', 'new line');
$m->prepend('log.txt', 'first line');

// Directory operations
$m->files('dir');                    // فقط سطح اول
$m->allFiles('dir');                  // بازگشتی
$m->directories('dir');
$m->allDirectories('dir');
$m->makeDirectory('new/dir');
$m->deleteDirectory('old/dir');

// Visibility
$m->setVisibility('file.txt', 'public');
$m->visibility('file.txt');          // 'public' یا 'private'

// Stream (برای فایل‌های بزرگ)
$stream = $m->readStream('video.mp4');
$m->writeStream('output.mp4', $stream);

// Disk جداگانه
$m->disk('public')->put('avatar.jpg', $content);
$m->disk('memory')->put('test.txt', 'content');

// Direct driver
$fs = new LocalFilesystem('/var/www/storage');
$fs->put('file.txt', 'content');

// Memory (برای تست)
$mem = new MemoryFilesystem();
$mem->put('test.txt', 'content');
```
